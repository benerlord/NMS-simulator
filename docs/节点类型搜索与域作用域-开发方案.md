```sql
CREATE TABLE IF NOT EXISTS domains (
  id          TEXT PRIMARY KEY,
  name        TEXT NOT NULL UNIQUE,
  description TEXT,
  created_at  TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at  TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS domain_node_types (
  domain_id     TEXT NOT NULL,
  node_type_id  TEXT NOT NULL,
  created_at    TEXT NOT NULL DEFAULT (datetime('now')),
  PRIMARY KEY (domain_id, node_type_id),
  FOREIGN KEY (domain_id) REFERENCES domains(id) ON DELETE CASCADE,
  FOREIGN KEY (node_type_id) REFERENCES node_types(id) ON DELETE CASCADE
);

CREATE INDEX IF NOT EXISTS idx_domain_node_types_domain ON domain_node_types(domain_id);
CREATE INDEX IF NOT EXISTS idx_domain_node_types_type ON domain_node_types(node_type_id);
```

## 3. 后端路由

### 3.1 新增 `admin/domain.py`

**文件**：`backend/app/admin/domain.py`（新建）

```python
import uuid
from fastapi import APIRouter, HTTPException
from app.db.connection import connect, transaction
from app.admin.schemas.domain import DomainCreate, DomainUpdate, DomainItem

router = APIRouter(prefix="/admin/api", tags=["domain"])

def _new_id() -> str:
    return f"dom_{uuid.uuid4().hex[:12]}"


@router.get("/domains")
def list_domains() -> dict:
    """列出所有域，含每个域下的拓扑数量"""
    with connect() as conn:
        rows = conn.execute("""
            SELECT d.*, COUNT(t.id) AS topology_count
            FROM domains d
            LEFT JOIN topologies t ON t.domain_id = d.id
            GROUP BY d.id
            ORDER BY d.name
        """).fetchall()
        items = [DomainItem(
            id=r["id"], name=r["name"], description=r["description"],
            topology_count=r["topology_count"],
            created_at=r["created_at"], updated_at=r["updated_at"],
        ).model_dump(mode="json", by_alias=True) for r in rows]
    return {"code": 0, "data": {"items": items, "total": len(items)}, "message": "ok"}


@router.post("/domains")
def create_domain(data: DomainCreate) -> dict:
    with transaction() as conn:
        existing = conn.execute(
            "SELECT id FROM domains WHERE name = ?", (data.name,)
        ).fetchone()
        if existing:
            raise HTTPException(
                status_code=409,
                detail={"code": 40301, "message": "域名称已存在"},
            )
        dom_id = _new_id()
        conn.execute(
            "INSERT INTO domains (id, name, description) VALUES (?, ?, ?)",
            (dom_id, data.name, data.description),
        )
    return {"code": 0, "data": {"id": dom_id}, "message": "ok"}


@router.put("/domains/{dom_id}")
def update_domain(dom_id: str, data: DomainUpdate) -> dict:
    with transaction() as conn:
        existing = conn.execute(
            "SELECT id FROM domains WHERE id = ?", (dom_id,)
        ).fetchone()
        if not existing:
            raise HTTPException(
                status_code=404,
                detail={"code": 40302, "message": "域不存在"},
            )
        if data.name is not None:
            dup = conn.execute(
                "SELECT id FROM domains WHERE name = ? AND id != ?",
                (data.name, dom_id),
            ).fetchone()
            if dup:
                raise HTTPException(
                    status_code=409,
                    detail={"code": 40301, "message": "域名称已存在"},
                )
        updates = {}
        if data.name is not None:
            updates["name"] = data.name
        if data.description is not None:
            updates["description"] = data.description
        if updates:
            updates["updated_at"] = "datetime('now')"
            set_clause = ", ".join(f"{k} = ?" for k in updates.keys())
            conn.execute(
                f"UPDATE domains SET {set_clause} WHERE id = ?",
                (*updates.values(), dom_id),
            )
    return {"code": 0, "data": {"id": dom_id}, "message": "ok"}


@router.delete("/domains/{dom_id}")
def delete_domain(dom_id: str) -> dict:
    with transaction() as conn:
        existing = conn.execute(
            "SELECT id FROM domains WHERE id = ?", (dom_id,)
        ).fetchone()
        if not existing:
            raise HTTPException(
                status_code=404,
                detail={"code": 40302, "message": "域不存在"},
            )
        # 级联：domain_node_types 自动由 FK ON DELETE CASCADE 清理
        # topologies.domain_id 置 NULL（保持拓扑存在）
        conn.execute(
            "UPDATE topologies SET domain_id = NULL WHERE domain_id = ?",
            (dom_id,),
        )
        conn.execute("DELETE FROM domains WHERE id = ?", (dom_id,))
    return {"code": 0, "data": {"id": dom_id}, "message": "ok"}
```

```python
@router.get("/node-types")
def list_node_types(domain_id: str = None) -> dict:
    with connect() as conn:
        if domain_id:
            # 检查该域是否有绑定记录
            has_bindings = conn.execute(
                "SELECT COUNT(*) as cnt FROM domain_node_types WHERE domain_id = ?",
                (domain_id,),
            ).fetchone()["cnt"] > 0
            if has_bindings:
                # 白名单模式：只返回绑定的类型
                rows = conn.execute("""
                    SELECT nt.* FROM node_types nt
                    INNER JOIN domain_node_types dnt ON dnt.node_type_id = nt.id
                    WHERE dnt.domain_id = ?
                    ORDER BY nt.category, nt.name
                """, (domain_id,)).fetchall()
            else:
                # 默认模式：该域未配置，返回全部
                rows = conn.execute(
                    "SELECT * FROM node_types ORDER BY category, name"
                ).fetchall()
        else:
            rows = conn.execute(
                "SELECT * FROM node_types ORDER BY category, name"
            ).fetchall()

        items = []
        for r in rows:
            # 查询该类型关联的域
            dom_rows = conn.execute("""
                SELECT d.id, d.name FROM domains d
                INNER JOIN domain_node_types dnt ON dnt.domain_id = d.id
                WHERE dnt.node_type_id = ?
            """, (r["id"],)).fetchall()

            item = NodeTypeDetail(
                # ... 现有字段不变 ...
                domain_ids=[dr["id"] for dr in dom_rows],
                domain_names=[dr["name"] for dr in dom_rows],
            )
            items.append(item.model_dump(mode="json", by_alias=True))
        return {"code": 0, "data": {"items": items, "total": len(items)}, "message": "ok"}
```

#### 3.2.2 新增单个/批量域关联端点

```python
@router.put("/node-types/{type_id}/domains")
def update_node_type_domains(type_id: str, data: NodeTypeDomainsUpdate) -> dict:
    """单个节点类型设置所属域（全量替换）"""
    with transaction() as conn:
        existing = conn.execute(
            "SELECT id FROM node_types WHERE id = ?", (type_id,)
        ).fetchone()
        if not existing:
            raise HTTPException(status_code=404, detail={"code": 40201, "message": "类型不存在"})
        conn.execute("DELETE FROM domain_node_types WHERE node_type_id = ?", (type_id,))
        for dom_id in data.domain_ids:
            conn.execute(
                "INSERT OR IGNORE INTO domain_node_types (domain_id, node_type_id) VALUES (?, ?)",
                (dom_id, type_id),
            )
    return {"code": 0, "data": {"id": type_id}, "message": "ok"}


@router.put("/node-types/domains")
def batch_update_node_type_domains(data: NodeTypeBatchDomainsUpdate) -> dict:
    """批量节点类型设置所属域（全量替换，domainIds: [] = 解除所有关联）"""
    with transaction() as conn:
        for type_id in data.node_type_ids:
            existing = conn.execute(
                "SELECT id FROM node_types WHERE id = ?", (type_id,)
            ).fetchone()
            if not existing:
                continue
            conn.execute("DELETE FROM domain_node_types WHERE node_type_id = ?", (type_id,))
            for dom_id in data.domain_ids:
                conn.execute(
                    "INSERT OR IGNORE INTO domain_node_types (domain_id, node_type_id) VALUES (?, ?)",
                    (dom_id, type_id),
                )
    return {
        "code": 0,
        "data": {"nodeTypeIds": data.node_type_ids, "domainIds": data.domain_ids},
        "message": "ok",
    }
```

#### 5.1.1 分类折叠 + 独立滚动

**问题**：节点类型数量多时，整个画布面板需要大量滚动才能浏览所有分类。单个分类下也可能有几十个类型。

**方案**：
- **分类标题可折叠** — 点击分类行切换折叠/展开，`▶` / `▼` 指示状态，默认全部展开
- **分类内容区独立滚动** — `.category-body` 设 `max-height: 240px` + `overflow-y: auto`，单个分类内类型再多也只在该区域内滚动，不撑长整个面板

```html
<div class="category-label" @click="toggleCategory(category)">
  <span class="category-arrow">{{ collapsed ? '▶' : '▼' }}</span>
  {{ label }} ({{ count }})
</div>
<div v-show="!collapsed" class="category-body">  <!-- max-height: 240px; overflow-y: auto -->
  <div v-for="nt in types" ...> ... </div>
</div>
```

```typescript
const collapsedCategories = ref(new Set<string>())
function toggleCategory(cat: string) {
  collapsedCategories.value.has(cat)
    ? collapsedCategories.value.delete(cat)
    : collapsedCategories.value.add(cat)
}
```

#### 5.3.3 批量操作处理函数

```typescript
const domainModalVisible = ref(false)
const domainIdsForBatch = ref<string[]>([])
const selectedRowKeys = ref<string[]>([])

// 是否有选中项已关联域（控制"解除关联"是否可选）
const hasAssociatedTypes = computed(() => {
  return filteredNodeTypes.value
    .filter(nt => selectedRowKeys.value.includes(nt.id))
    .some(nt => nt.domainIds.length > 0)
})

async function handleBatchMenuClick({ key }: { key: string }) {
  switch (key) {
    case 'associate-domain':
      domainIdsForBatch.value = []
      domainModalVisible.value = true
      break
    case 'unbind-domain':
      Modal.confirm({
        title: `确认解除 ${selectedRowKeys.value.length} 个节点类型的域关联？`,
        content: '解除后这些类型将变为全局可用',
        onOk: async () => {
          await nodeTypeApi.batchUpdateDomains(selectedRowKeys.value, [])
          message.success('已解除关联')
          fetchNodeTypes()
        },
      })
      break
    case 'batch-delete':
      await handleBatchDelete()
      break
  }
}

async function handleBatchAssociateDomains() {
  await nodeTypeApi.batchUpdateDomains(selectedRowKeys.value, domainIdsForBatch.value)
  message.success(`已将 ${selectedRowKeys.value.length} 个类型关联到 ${domainIdsForBatch.value.length} 个域`)
  domainModalVisible.value = false
  selectedRowKeys.value = []
  fetchNodeTypes()
}

// 单个解除域关联（点击 Tag x）
async function removeDomain(typeId: string, domainId: string) {
  const nt = filteredNodeTypes.value.find(t => t.id === typeId)
  if (!nt) return
  const newDomains = nt.domainIds.filter(id => id !== domainId)
  await nodeTypeApi.updateDomains(typeId, newDomains)
  fetchNodeTypes()
}
```

```html
    </Select.Option>
  </Select>
</Form.Item>
```

拓扑创建/编辑时需要加载域列表（通过 props 或 composable 传入）。
