```python
@router.post("/apis/export")
def export_apis(data: ApiExportRequest) -> dict:
    """导出接口为 JSON 文件"""
    with connect() as conn:
        if data.ids:
            placeholders = ",".join("?" for _ in data.ids)
            rows = conn.execute(
                f"SELECT * FROM api_configs WHERE id IN ({placeholders}) ORDER BY method, path",
                tuple(data.ids),
            ).fetchall()
        elif data.domain_id:
            rows = conn.execute(
                "SELECT * FROM api_configs WHERE domain_id = ? ORDER BY method, path",
                (data.domain_id,),
            ).fetchall()
        else:
            rows = conn.execute(
                "SELECT * FROM api_configs ORDER BY method, path"
            ).fetchall()
        apis = [dict(r) for r in rows]
    doc = ApiExportDoc(
        exported_at=datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%SZ"),
        apis=apis,
    )
    return {
        "code": 0,
        "data": doc.model_dump(mode="json", by_alias=True),
        "message": "ok",
    }

@router.post("/apis/import")
def import_apis(file: UploadFile = File(...)) -> dict:
    """导入 JSON 文件，以 (method, path) 匹配新建或覆盖"""
    if not file.filename or not file.filename.endswith('.json'):
        raise HTTPException(400, detail={"code": 40410, "message": "仅支持 .json 文件"})

    contents = await file.read()
    try:
        doc = json.loads(contents)
    except json.JSONDecodeError:
        raise HTTPException(400, detail={"code": 40411, "message": "JSON 解析失败"})

    apis = doc.get("apis", [])
    if not isinstance(apis, list):
        raise HTTPException(400, detail={"code": 40412, "message": "格式无效：缺少 apis 数组"})

    result = ApiImportResult()
    with transaction() as conn:
        for api_data in apis:
            method = api_data.get("method")
            path = api_data.get("path")
            instance_id = api_data.get("instanceId")
            if not method or not path:
                result.errors.append(f"缺少 method/path，跳过")
                continue

            # 以 domain_id + method + path 匹配
            existing = conn.execute(
                "SELECT id FROM api_configs WHERE domain_id IS ? AND method = ? AND path = ?",
                (api_data.get("domainId"), method, path),
            ).fetchone()

            if existing:
                # 覆盖更新
                _update_api_row(conn, existing["id"], api_data)
                result.updated += 1
            else:
                # 新建
                _insert_api_row(conn, api_data)
                result.created += 1

    return {
        "code": 0,
        "data": result.model_dump(mode="json", by_alias=True),
        "message": "ok",
    }
```

## 问题 2：SQL 列字段格式统一

### 2.1 根因

- `SqlViewPanel` 从 SQLite 视图读取列名，均为 `snake_case`
- `ParamMappingTable` 的 `bindTo` 字段由用户手动输入，格式无约束
- `ApiConfigModal` 的 `availableFieldNames` 来自请求校验字段名（用户自定义），非 SQL 列名
- 用户不清楚 `bindTo` 应该用哪种格式

### 2.2 方案

**不改后端**，纯前端优化。核心思路：`availableFieldNames` 改为从 SQL 测试运行结果中自动提取列名。

**文件**：`frontend/src/components/apis/ApiConfigModal.vue`

```typescript
// 新增：SQL 测试运行后提取列名
const sqlColumnNames = ref<string[]>([])
const columnNamesTip = 'SQL 列名使用 snake_case 格式'

function onSqlTestResult(result: { columns: string[]; rows: any[] }) {
  sqlColumnNames.value = result.columns  // 自动提取 snake_case 列名
}

// availableFieldNames 合并：SQL 列名 + 请求参数字段名
const availableFieldNames = computed(() => {
  const names = new Set<string>()
  // SQL 列名（snake_case）
  for (const col of sqlColumnNames.value) names.add(col)
  // 请求校验的参数字段名
  for (const q of formState.value.requestQuery) {
    if (q.name.trim()) names.add(q.name.trim())
  }
  return [...names]
})
```

**文件**：`frontend/src/components/apis/ParamMappingTable.vue`

`bindTo` 列的 AutoComplete 下拉追加格式提示：

```html
<AutoComplete
  :value="record.bindTo"
  :options="[
    ...availableFieldNames.map(n => ({ label: `📎 ${n}`, value: n })),
  ]"
  placeholder="绑定 SQL 列名（snake_case）"
/>
```

列名旁加 Tooltip 提示：`"列名格式来自数据库，使用 snake_case"`。

### 2.3 实施

单一改动点：`ApiConfigModal.vue` + `SqlRunner.vue`（让测试结果输出列名列表）+ `ParamMappingTable.vue`。前端 3 个文件。

## 问题 4：多端口实例

### 设计说明

**关联链路**：实例 → 拓扑 → 网管/设备。实例绑定拓扑（而非直接绑定域），自动继承拓扑下所有接口。无需逐接口绑定，无需修改 `api_configs` 表。

**为何不需要 `api_configs.instance_id`**：不同实例 = 不同端口 = 独立 FastAPI 子进程，各自注册路由表。`GET /api/health:8081` 与 `GET /api/health:9081` 是不同端口的端点，数据库不存在冲突，UNIQUE 约束无需修改。

#### 4.8.1 数据库

| 值 | 含义 |
|------|------|
| `running` | 子进程正常运行 |
| `stopped` | enabled=false，子进程未启动 |
| `error` | 启动失败或反复崩溃 |
| `starting` | 启动中（瞬态） |

#### 4.8.2 InstanceRunner — 进程管理器

**文件**：`backend/app/core/instance_runner.py`（新建）

```python
import subprocess
import time
from threading import Lock

class InstanceRunner:
    def __init__(self):
        self._processes: dict[str, subprocess.Popen] = {}
        self._restart_counts: dict[str, list[float]] = {}  # inst_id -> restart timestamps
        self._lock = Lock()

    def sync_all(self):
        """主进程启动时调用：为所有 enabled=1 的实例启动子进程"""
        with connect() as conn:
            rows = conn.execute(
                "SELECT id, port, topology_id FROM mock_instances WHERE enabled = 1"
            ).fetchall()
        for r in rows:
            self.start_instance(r["id"], r["port"], r["topology_id"])

    def start_instance(self, inst_id: str, port: int, topology_id: str):
        """启动 uvicorn 子进程，仅加载指定拓扑的接口"""
        with self._lock:
            if inst_id in self._processes:
                return
            _update_status(inst_id, "starting")
            try:
                proc = subprocess.Popen([
                    "python", "-m", "app.mock.instance_app",
                    "--topology-id", topology_id,
                    "--port", str(port),
                ])
                self._processes[inst_id] = proc
                _update_status(inst_id, "running")
            except Exception:
                _update_status(inst_id, "error")

    def stop_instance(self, inst_id: str):
        with self._lock:
            proc = self._processes.pop(inst_id, None)
            if proc:
                proc.terminate()
                proc.wait(timeout=5)
            _update_status(inst_id, "stopped")

    def restart_instance(self, inst_id: str, port: int, topology_id: str):
        self.stop_instance(inst_id)
        self.start_instance(inst_id, port, topology_id)

    def shutdown_all(self):
        for inst_id in list(self._processes.keys()):
            self.stop_instance(inst_id)

    def check_and_restart(self):
        """定期健康检查：检测已崩溃子进程，自动重启（限流）"""
        for inst_id, proc in list(self._processes.items()):
            if proc.poll() is not None:  # 进程已退出
                # 限流：1 分钟内重启超过 3 次则标记 error
                now = time.time()
                self._restart_counts.setdefault(inst_id, []).append(now)
                self._restart_counts[inst_id] = [
                    t for t in self._restart_counts[inst_id] if now - t < 60
                ]
                if len(self._restart_counts[inst_id]) > 3:
                    _update_status(inst_id, "error")
                    self._processes.pop(inst_id, None)
                    continue
                # 重新启动
                self.stop_instance(inst_id)
                # 从 DB 读取最新配置
                with connect() as conn:
                    row = conn.execute(
                        "SELECT port, topology_id FROM mock_instances WHERE id = ?",
                        (inst_id,),
                    ).fetchone()
                if row:
                    self.start_instance(inst_id, row["port"], row["topology_id"])
```

```python
"""
每个实例子进程的入口模块。
用法: python -m app.mock.instance_app --topology-id topo_xxx --port 8081

import argparse
import uvicorn
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from app.db.connection import connect, init_db
from app.mock.handler import make_handler


def create_app(topology_id: str) -> FastAPI:
    app = FastAPI(title=f"Mock Instance [{topology_id}]")

    app.add_middleware(
        CORSMiddleware,
        allow_origins=["*"],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    with connect() as conn:
        rows = conn.execute(
            "SELECT id, method, path FROM api_configs WHERE topology_id = ? AND enabled = 1",
            (topology_id,),
        ).fetchall()

    for row in rows:
        handler = make_handler(row["id"])
        app.add_api_route(
            path=row["path"],
            endpoint=handler,
            methods=[row["method"]],
        )

    return app


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--topology-id", required=True)
    parser.add_argument("--port", type=int, required=True)
    args = parser.parse_args()

    init_db()
    app = create_app(args.topology_id)
    uvicorn.run(app, host="0.0.0.0", port=args.port, log_level="info")
```

```python
    mock_registry.bind(app)
    mock_registry.load_all()

    global runner
    runner = InstanceRunner()
    runner.sync_all()

    yield

    if runner:
        runner.shutdown_all()
```

**文件**：`backend/app/admin/mock_instance.py` CRUD 端点中触发 Runner：

- `create_mock_instance` — 如果 `enabled=True`，`runner.start_instance(inst_id, port, topology_id)`
- `delete_mock_instance` — `runner.stop_instance(inst_id)`
- `update_mock_instance` — 如果 port 或 topology 变化，`runner.restart_instance(inst_id, ...)`
- `patch_instance_enabled` — false→true: start; true→false: stop
